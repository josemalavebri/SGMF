CREATE OR ALTER PROCEDURE dbo.usp_Movimiento_Create
    @IdTipoMovimiento INT,
    @IdTipoIngreso INT = NULL,
    @IdCategoriaEgreso INT = NULL,
    @Monto DECIMAL(12, 2),
    @Descripcion VARCHAR(200)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @IdIngreso INT;
    DECLARE @IdEgreso INT;

    SELECT @IdIngreso = IdTipoMovimiento
    FROM dbo.TipoMovimiento
    WHERE Nombre = 'Ingreso';

    SELECT @IdEgreso = IdTipoMovimiento
    FROM dbo.TipoMovimiento
    WHERE Nombre = 'Egreso';

    IF @IdIngreso IS NULL OR @IdEgreso IS NULL
        THROW 50001, 'No están configurados los tipos Ingreso y Egreso.', 1;

    IF @IdTipoMovimiento NOT IN (@IdIngreso, @IdEgreso)
        THROW 50002, 'El tipo de movimiento no es válido.', 1;

    IF @Monto IS NULL OR @Monto <= 0
        THROW 50003, 'El monto debe ser mayor que cero.', 1;

    IF @Descripcion IS NULL OR LEN(LTRIM(RTRIM(@Descripcion))) = 0
        THROW 50004, 'La descripción es obligatoria.', 1;

    IF @IdTipoMovimiento = @IdIngreso
    BEGIN
        IF @IdTipoIngreso IS NULL OR @IdCategoriaEgreso IS NOT NULL
            THROW 50005, 'Un ingreso requiere un tipo de ingreso y no admite categoría de egreso.', 1;

        IF NOT EXISTS (
            SELECT 1
            FROM dbo.TipoIngreso
            WHERE IdTipoIngreso = @IdTipoIngreso
              AND Activo = 1
        )
            THROW 50006, 'El tipo de ingreso no existe o está inactivo.', 1;
    END
    ELSE
    BEGIN
        IF @IdCategoriaEgreso IS NULL OR @IdTipoIngreso IS NOT NULL
            THROW 50007, 'Un egreso requiere una categoría y no admite tipo de ingreso.', 1;

        IF NOT EXISTS (
            SELECT 1
            FROM dbo.CategoriaEgreso
            WHERE IdCategoriaEgreso = @IdCategoriaEgreso
              AND Activo = 1
        )
            THROW 50008, 'La categoría de egreso no existe o está inactiva.', 1;
    END;

    INSERT INTO dbo.Movimiento (
        IdTipoMovimiento,
        IdTipoIngreso,
        IdCategoriaEgreso,
        Monto,
        Descripcion
    )
    VALUES (
        @IdTipoMovimiento,
        @IdTipoIngreso,
        @IdCategoriaEgreso,
        @Monto,
        @Descripcion
    );

    SELECT CAST(SCOPE_IDENTITY() AS INT) AS IdMovimiento;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Movimiento_Update
    @IdMovimiento INT,
    @IdTipoMovimiento INT,
    @IdTipoIngreso INT = NULL,
    @IdCategoriaEgreso INT = NULL,
    @Monto DECIMAL(12, 2),
    @Descripcion VARCHAR(200)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @IdIngreso INT;
    DECLARE @IdEgreso INT;
    DECLARE @TipoIngresoActual INT;
    DECLARE @CategoriaActual INT;

    SELECT @IdIngreso = IdTipoMovimiento
    FROM dbo.TipoMovimiento
    WHERE Nombre = 'Ingreso';

    SELECT @IdEgreso = IdTipoMovimiento
    FROM dbo.TipoMovimiento
    WHERE Nombre = 'Egreso';

    IF @IdIngreso IS NULL OR @IdEgreso IS NULL
        THROW 50101, 'No están configurados los tipos Ingreso y Egreso.', 1;

    IF @IdTipoMovimiento NOT IN (@IdIngreso, @IdEgreso)
        THROW 50102, 'El tipo de movimiento no es válido.', 1;

    IF @Monto IS NULL OR @Monto <= 0
        THROW 50103, 'El monto debe ser mayor que cero.', 1;

    IF @Descripcion IS NULL OR LEN(LTRIM(RTRIM(@Descripcion))) = 0
        THROW 50104, 'La descripción es obligatoria.', 1;

    SELECT
        @TipoIngresoActual = IdTipoIngreso,
        @CategoriaActual = IdCategoriaEgreso
    FROM dbo.Movimiento
    WHERE IdMovimiento = @IdMovimiento;

    IF @@ROWCOUNT = 0
        THROW 50105, 'El movimiento no existe.', 1;

    IF @IdTipoMovimiento = @IdIngreso
    BEGIN
        IF @IdTipoIngreso IS NULL OR @IdCategoriaEgreso IS NOT NULL
            THROW 50106, 'Un ingreso requiere un tipo de ingreso y no admite categoría de egreso.', 1;

        IF NOT EXISTS (
            SELECT 1
            FROM dbo.TipoIngreso
            WHERE IdTipoIngreso = @IdTipoIngreso
              AND (
                  Activo = 1
                  OR @IdTipoIngreso = @TipoIngresoActual
              )
        )
            THROW 50107, 'El tipo de ingreso no existe o está inactivo.', 1;
    END
    ELSE
    BEGIN
        IF @IdCategoriaEgreso IS NULL OR @IdTipoIngreso IS NOT NULL
            THROW 50108, 'Un egreso requiere una categoría y no admite tipo de ingreso.', 1;

        IF NOT EXISTS (
            SELECT 1
            FROM dbo.CategoriaEgreso
            WHERE IdCategoriaEgreso = @IdCategoriaEgreso
              AND (
                  Activo = 1
                  OR @IdCategoriaEgreso = @CategoriaActual
              )
        )
            THROW 50109, 'La categoría de egreso no existe o está inactiva.', 1;
    END;

    UPDATE dbo.Movimiento
    SET
        IdTipoMovimiento = @IdTipoMovimiento,
        IdTipoIngreso = @IdTipoIngreso,
        IdCategoriaEgreso = @IdCategoriaEgreso,
        Monto = @Monto,
        Descripcion = @Descripcion
    WHERE IdMovimiento = @IdMovimiento;
END;
GO

CREATE OR ALTER PROCEDURE usp_Reporte_Balance
AS
BEGIN
  SET NOCOUNT ON;
  SELECT ISNULL(SUM(CASE WHEN IdTipoMovimiento = 1 THEN Monto ELSE 0 END), 0) AS TotalIngresos,
         ISNULL(SUM(CASE WHEN IdTipoMovimiento = 2 THEN Monto ELSE 0 END), 0) AS TotalEgresos,
         ISNULL(SUM(CASE WHEN IdTipoMovimiento = 1 THEN Monto WHEN IdTipoMovimiento = 2 THEN -Monto ELSE 0 END), 0) AS Saldo
  FROM   Movimiento;
END
GO

CREATE OR ALTER PROCEDURE usp_Reporte_MovimientosPorFecha
@FechaInicio DATETIME2, @FechaFin DATETIME2
AS
BEGIN
  SET NOCOUNT ON;
  SELECT   m.IdMovimiento,
           tm.Nombre AS TipoMovimiento,
           ti.Nombre AS TipoIngreso,
           ce.Nombre AS CategoriaEgreso,
           m.Monto,
           m.Descripcion,
           m.Fecha
  FROM     Movimiento AS m
           INNER JOIN
           TipoMovimiento AS tm
           ON m.IdTipoMovimiento = tm.IdTipoMovimiento
           LEFT OUTER JOIN
           TipoIngreso AS ti
           ON m.IdTipoIngreso = ti.IdTipoIngreso
           LEFT OUTER JOIN
           CategoriaEgreso AS ce
           ON m.IdCategoriaEgreso = ce.IdCategoriaEgreso
  WHERE    m.Fecha >= @FechaInicio
           AND m.Fecha < @FechaFin
  ORDER BY m.Fecha DESC, m.IdMovimiento DESC;
END
GO

USE SGMF;
GO

CREATE TABLE TipoMovimiento (
    IdTipoMovimiento INT IDENTITY (1, 1) PRIMARY KEY,
    Nombre VARCHAR (20) NOT NULL UNIQUE
);
GO

CREATE TABLE TipoIngreso (
    IdTipoIngreso INT IDENTITY (1, 1) PRIMARY KEY,
    Nombre VARCHAR (50) NOT NULL UNIQUE,
    Activo BIT DEFAULT 1 NOT NULL
);
GO

CREATE TABLE CategoriaEgreso (
    IdCategoriaEgreso INT IDENTITY (1, 1) PRIMARY KEY,
    Nombre VARCHAR (50) NOT NULL UNIQUE,
    Activo BIT DEFAULT 1 NOT NULL
);
GO

CREATE TABLE Movimiento (
    IdMovimiento INT IDENTITY (1, 1) PRIMARY KEY,
    IdTipoMovimiento INT NOT NULL,
    IdTipoIngreso INT NULL,
    IdCategoriaEgreso INT NULL,
    Monto DECIMAL (12, 2) NOT NULL,
    Descripcion VARCHAR (200) NOT NULL,
    Fecha DATETIME2 DEFAULT SYSDATETIME() NOT NULL,
    CONSTRAINT FK_Movimiento_TipoMovimiento
        FOREIGN KEY (IdTipoMovimiento)
        REFERENCES TipoMovimiento (IdTipoMovimiento),
    CONSTRAINT FK_Movimiento_TipoIngreso
        FOREIGN KEY (IdTipoIngreso)
        REFERENCES TipoIngreso (IdTipoIngreso),
    CONSTRAINT FK_Movimiento_CategoriaEgreso
        FOREIGN KEY (IdCategoriaEgreso)
        REFERENCES CategoriaEgreso (IdCategoriaEgreso)
);
GO
